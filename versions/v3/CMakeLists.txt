cmake_minimum_required(VERSION 3.16)
project(rmtc3_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rmtc3_core src/rmtc2/types.cpp src/rmtc2/grid.cpp src/rmtc2/parse.cpp src/rmtc2/bot.cpp)
target_include_directories(rmtc3_core PUBLIC src third_party/nlohmann/include)
if(MSVC)
 target_compile_options(rmtc3_core PUBLIC /utf-8 /W4)
else()
 target_compile_options(rmtc3_core PUBLIC -Wall -Wextra)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests rmtc3_core)
add_test(NAME core_regression COMMAND core_tests)

find_package(Threads REQUIRED)
add_executable(llm_worker_tests tests/llm_worker_tests.cpp ros2_ws/src/rmtc_bot2/src/llm.cpp)
target_include_directories(llm_worker_tests PRIVATE ros2_ws/src/rmtc_bot2/include third_party/nlohmann/include)
target_link_libraries(llm_worker_tests PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(llm_worker_tests PRIVATE /utf-8 /W4)
endif()
add_test(NAME llm_session_lifecycle COMMAND llm_worker_tests)
