cmake_minimum_required(VERSION 3.16)
project(rmtc6_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rmtc6_core src/rmtc6/types.cpp src/rmtc6/grid.cpp src/rmtc6/parse.cpp src/rmtc6/bot.cpp src/rmtc6/events.cpp src/rmtc6/battle.cpp)
target_include_directories(rmtc6_core PUBLIC src third_party/nlohmann/include)
if(MSVC)
 target_compile_options(rmtc6_core PUBLIC /utf-8 /W4)
else()
 target_compile_options(rmtc6_core PUBLIC -Wall -Wextra)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests rmtc6_core)
add_test(NAME core_regression COMMAND core_tests)

find_package(Threads REQUIRED)
add_executable(llm_worker_tests tests/llm_worker_tests.cpp ros2_ws/src/rmtc_bot6/src/llm.cpp)
target_include_directories(llm_worker_tests PRIVATE ros2_ws/src/rmtc_bot6/include third_party/nlohmann/include)
target_link_libraries(llm_worker_tests PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(llm_worker_tests PRIVATE /utf-8 /W4)
endif()
add_test(NAME llm_session_lifecycle COMMAND llm_worker_tests)

add_executable(rmtc_replay tools/replay_main.cpp)
target_link_libraries(rmtc_replay PRIVATE rmtc6_core)
add_executable(v4_tests tests/v4_tests.cpp tools/bench/referee.cpp)
target_include_directories(v4_tests PRIVATE tools/bench)
target_link_libraries(v4_tests rmtc6_core)
add_test(NAME v4_rules_boundaries COMMAND v4_tests)
set_tests_properties(core_regression llm_session_lifecycle v4_rules_boundaries PROPERTIES TIMEOUT 30)

# Optional benchmark compiles the untouched v4 and v5 sources as opponents.
option(RMTC_BUILD_BENCH "Build v6/v5/v4 comparison (needs sibling source folders)" OFF)
set(RMTC_V4_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../校内4.0" CACHE PATH "Unmodified v4 source")
set(RMTC_V5_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../校内5.0" CACHE PATH "Unmodified v5 source")
if(RMTC_BUILD_BENCH)
 foreach(ver IN ITEMS 4 5)
  if(ver EQUAL 4)
   set(ns rmtc2)
  else()
   set(ns rmtc3)
  endif()
  set(base "${RMTC_V${ver}_ROOT}")
  if(NOT EXISTS "${base}/src/${ns}/bot.cpp")
   message(FATAL_ERROR "Missing unchanged v${ver} sources at ${base}")
  endif()
  add_library(baseline${ver} STATIC "${base}/src/${ns}/types.cpp" "${base}/src/${ns}/grid.cpp" "${base}/src/${ns}/bot.cpp" "${base}/src/${ns}/parse.cpp" "${base}/src/${ns}/events.cpp")
  target_include_directories(baseline${ver} PUBLIC "${base}/src" "${base}/third_party/nlohmann/include")
  if(MSVC)
   target_compile_options(baseline${ver} PUBLIC /utf-8)
  endif()
 endforeach()
 add_executable(rmtc_arena tools/bench/bench_main.cpp tools/bench/referee.cpp tools/bench/opponent_adapter.cpp tools/bench/v5_adapter.cpp)
 target_include_directories(rmtc_arena PRIVATE tools/bench)
 target_link_libraries(rmtc_arena rmtc6_core baseline4 baseline5)
endif()
