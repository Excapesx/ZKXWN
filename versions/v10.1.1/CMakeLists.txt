cmake_minimum_required(VERSION 3.16)
project(rmtc9_core VERSION 10.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(rmtc9_core src/rmtc9/types.cpp src/rmtc9/grid.cpp src/rmtc9/parse.cpp src/rmtc9/bot.cpp src/rmtc9/events.cpp src/rmtc9/battle.cpp)
target_include_directories(rmtc9_core PUBLIC src third_party/nlohmann/include)
if(MSVC)
  target_compile_options(rmtc9_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(rmtc9_core PUBLIC -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)

  target_include_directories(test_referee PUBLIC tests/support)
  target_link_libraries(test_referee PUBLIC rmtc9_core)
  function(core_test target name referee)
    add_executable(${target} tests/${target}.cpp)
    target_link_libraries(${target} PRIVATE rmtc9_core)
    if(referee)
      target_link_libraries(${target} PRIVATE test_referee)
    endif()
    add_test(NAME ${name} COMMAND ${target})
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endfunction()
