cmake_minimum_required(VERSION 3.16)
project(v1011_selection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
get_filename_component(VERSION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(BENCH_ROOT "${REPO_ROOT}/versions/v10.2")
add_library(candidate STATIC)
foreach(unit types grid parse bot events battle)
  target_sources(candidate PRIVATE "${VERSION_ROOT}/src/rmtc9/${unit}.cpp")
endforeach()
target_include_directories(candidate PUBLIC "${VERSION_ROOT}/src" "${VERSION_ROOT}/third_party/nlohmann/include")
foreach(ver 7 8 91 93 100 101)
  add_library(frozen${ver} STATIC)
  foreach(unit types grid parse bot events battle)
    target_sources(frozen${ver} PRIVATE "${BENCH_ROOT}/baseline_v${ver}/src/rmtc${ver}/${unit}.cpp")
  endforeach()
  target_include_directories(frozen${ver} PUBLIC "${BENCH_ROOT}/baseline_v${ver}/src" "${VERSION_ROOT}/third_party/nlohmann/include")
endforeach()
add_executable(selection_arena arena_main.cpp "${BENCH_ROOT}/tools/bench/referee.cpp")
foreach(ver 7 8 91 93 100 101)
  target_sources(selection_arena PRIVATE "${BENCH_ROOT}/tools/bench/v${ver}_adapter.cpp")
endforeach()
target_include_directories(selection_arena PRIVATE "${BENCH_ROOT}/tools/bench")
target_link_libraries(selection_arena PRIVATE candidate frozen7 frozen8 frozen91 frozen93 frozen100 frozen101 Threads::Threads)
