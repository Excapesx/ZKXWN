cmake_minimum_required(VERSION 3.16)
project(rmtc4_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rmtc4_core src/rmtc2/types.cpp src/rmtc2/grid.cpp src/rmtc2/parse.cpp src/rmtc2/bot.cpp src/rmtc2/events.cpp)
target_include_directories(rmtc4_core PUBLIC src third_party/nlohmann/include)
if(MSVC)
 target_compile_options(rmtc4_core PUBLIC /utf-8 /W4)
else()
 target_compile_options(rmtc4_core PUBLIC -Wall -Wextra)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests rmtc4_core)
add_test(NAME core_regression COMMAND core_tests)

find_package(Threads REQUIRED)
add_executable(llm_worker_tests tests/llm_worker_tests.cpp ros2_ws/src/rmtc_bot2/src/llm.cpp)
target_include_directories(llm_worker_tests PRIVATE ros2_ws/src/rmtc_bot2/include third_party/nlohmann/include)
target_link_libraries(llm_worker_tests PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(llm_worker_tests PRIVATE /utf-8 /W4)
endif()
add_test(NAME llm_session_lifecycle COMMAND llm_worker_tests)

add_executable(rmtc_replay tools/replay_main.cpp)
target_link_libraries(rmtc_replay PRIVATE rmtc4_core)
add_executable(v4_tests tests/v4_tests.cpp tools/bench/referee.cpp)
target_include_directories(v4_tests PRIVATE tools/bench)
target_link_libraries(v4_tests rmtc4_core)
add_test(NAME v4_rules_boundaries COMMAND v4_tests)
set_tests_properties(core_regression llm_session_lifecycle v4_rules_boundaries PROPERTIES TIMEOUT 30)

option(RMTC_BUILD_HTTP "Compile the production libcurl transport (requires CURL development package)" OFF)
if(RMTC_BUILD_HTTP)
 find_package(CURL REQUIRED)
 add_library(rmtc4_http ros2_ws/src/rmtc_bot2/src/curl_transport.cpp)
 target_include_directories(rmtc4_http PUBLIC ros2_ws/src/rmtc_bot2/include)
 target_link_libraries(rmtc4_http PUBLIC CURL::libcurl Threads::Threads)
endif()

add_executable(rmtc_smoke tools/smoke_main.cpp tools/bench/referee.cpp)
target_include_directories(rmtc_smoke PRIVATE tools/bench)
target_link_libraries(rmtc_smoke rmtc4_core)
