cmake_minimum_required(VERSION 3.16)
project(rmtc7_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rmtc7_core src/rmtc7/types.cpp src/rmtc7/grid.cpp src/rmtc7/parse.cpp src/rmtc7/bot.cpp src/rmtc7/events.cpp src/rmtc7/battle.cpp)
target_include_directories(rmtc7_core PUBLIC src third_party/nlohmann/include)
if(MSVC)
 target_compile_options(rmtc7_core PUBLIC /utf-8 /W4)
else()
 target_compile_options(rmtc7_core PUBLIC -Wall -Wextra)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests rmtc7_core)
add_test(NAME core_regression COMMAND core_tests)

if(NOT WIN32)
 find_package(Threads REQUIRED)
endif()
add_executable(llm_worker_tests tests/llm_worker_tests.cpp ros2_ws/src/rmtc_bot7/src/llm.cpp)
target_include_directories(llm_worker_tests PRIVATE ros2_ws/src/rmtc_bot7/include third_party/nlohmann/include)
if(NOT WIN32)
 target_link_libraries(llm_worker_tests PRIVATE Threads::Threads)
endif()
if(MSVC)
 target_compile_options(llm_worker_tests PRIVATE /utf-8 /W4)
endif()
add_test(NAME llm_session_lifecycle COMMAND llm_worker_tests)

add_executable(rmtc_replay tools/replay_main.cpp)
target_link_libraries(rmtc_replay PRIVATE rmtc7_core)
add_executable(v4_tests tests/v4_tests.cpp tools/bench/referee.cpp)
target_include_directories(v4_tests PRIVATE tools/bench)
target_link_libraries(v4_tests rmtc7_core)
add_test(NAME v4_rules_boundaries COMMAND v4_tests)
set_tests_properties(core_regression llm_session_lifecycle v4_rules_boundaries PROPERTIES TIMEOUT 30)

option(RMTC_BUILD_HTTP "Compile the production libcurl transport (requires CURL development package)" OFF)
if(RMTC_BUILD_HTTP)
 find_package(CURL REQUIRED)
 add_library(rmtc7_http ros2_ws/src/rmtc_bot7/src/curl_transport.cpp)
 target_include_directories(rmtc7_http PUBLIC ros2_ws/src/rmtc_bot7/include)
 target_link_libraries(rmtc7_http PUBLIC CURL::libcurl Threads::Threads)
endif()

add_executable(rmtc_smoke tools/smoke_main.cpp tools/bench/referee.cpp)
target_include_directories(rmtc_smoke PRIVATE tools/bench)
target_link_libraries(rmtc_smoke rmtc7_core)

# Optional benchmark: 7.0 versus the untouched 6.0 sources as the opponent.
option(RMTC_BUILD_BENCH "Build 7.0 vs 6.0 comparison (needs the sibling 6.0 source folder)" OFF)
set(RMTC_V6_ROOT "" CACHE PATH "Unmodified 6.0 source; auto-detected when empty")
if(NOT RMTC_V6_ROOT)
  file(GLOB _rmtc_v6_candidates "${CMAKE_CURRENT_SOURCE_DIR}/../*6.0")
  if(_rmtc_v6_candidates)
    list(GET _rmtc_v6_candidates 0 RMTC_V6_ROOT)
  endif()
endif()
if(RMTC_BUILD_BENCH)
 if(NOT EXISTS "${RMTC_V6_ROOT}/src/rmtc6/bot.cpp")
  message(FATAL_ERROR "Missing unchanged 6.0 sources at ${RMTC_V6_ROOT}")
 endif()
 add_library(baseline6 STATIC
   "${RMTC_V6_ROOT}/src/rmtc6/types.cpp"
   "${RMTC_V6_ROOT}/src/rmtc6/grid.cpp"
   "${RMTC_V6_ROOT}/src/rmtc6/bot.cpp"
   "${RMTC_V6_ROOT}/src/rmtc6/parse.cpp"
   "${RMTC_V6_ROOT}/src/rmtc6/events.cpp"
   "${RMTC_V6_ROOT}/src/rmtc6/battle.cpp")
 target_include_directories(baseline6 PUBLIC "${RMTC_V6_ROOT}/src" "${RMTC_V6_ROOT}/third_party/nlohmann/include")
 if(MSVC)
  target_compile_options(baseline6 PUBLIC /utf-8)
 endif()
 add_executable(rmtc_arena tools/bench/bench_main.cpp tools/bench/referee.cpp tools/bench/v6_adapter.cpp)
 target_include_directories(rmtc_arena PRIVATE tools/bench)
 target_link_libraries(rmtc_arena rmtc7_core baseline6)
endif()

add_executable(v6_tests tests/v6_tests.cpp tools/bench/referee.cpp)
target_include_directories(v6_tests PRIVATE tools/bench)
target_link_libraries(v6_tests rmtc7_core)
add_test(NAME v7_physical_scenarios COMMAND v6_tests)
set_tests_properties(v7_physical_scenarios PROPERTIES TIMEOUT 30)
