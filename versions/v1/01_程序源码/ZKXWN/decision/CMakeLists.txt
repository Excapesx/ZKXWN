cmake_minimum_required(VERSION 3.16)
project(rmtc_decision VERSION 0.2.0 LANGUAGES CXX)

add_library(rmtc_decision src/engine.cpp)
target_include_directories(rmtc_decision PUBLIC include)
target_compile_features(rmtc_decision PUBLIC cxx_std_17)
set_target_properties(rmtc_decision PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(rmtc_decision PRIVATE /W4 /utf-8)
else()
  target_compile_options(rmtc_decision PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(decision_demo examples/demo.cpp)
target_link_libraries(decision_demo PRIVATE rmtc_decision)
target_include_directories(decision_demo PRIVATE examples)

include(CTest)
if(BUILD_TESTING)
  add_executable(decision_tests tests/scenarios.cpp)
  target_link_libraries(decision_tests PRIVATE rmtc_decision)
  target_include_directories(decision_tests PRIVATE examples)
  add_test(NAME decision_scenarios COMMAND decision_tests)
endif()
