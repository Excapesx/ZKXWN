# SHARED
cmake_minimum_required(VERSION 3.16)
project(rmtc_bot VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RMTC_WITH_ROS "Build ROS 2 node (Ubuntu/Humble)" ON)
option(RMTC_WITH_CURL "Build real HTTP transport" ON)
include(CTest)
find_package(Threads REQUIRED)
get_filename_component(RMTC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT EXISTS "${RMTC_ROOT}/decision/CMakeLists.txt")
  message(FATAL_ERROR "Keep decision/, third_party/, rmtc_ws/ together at repository root")
endif()
add_subdirectory("${RMTC_ROOT}/decision" "${CMAKE_CURRENT_BINARY_DIR}/decision")
find_package(nlohmann_json QUIET)
if(NOT nlohmann_json_FOUND)
  add_library(rmtc_json INTERFACE)
  target_include_directories(rmtc_json INTERFACE "${RMTC_ROOT}/third_party/nlohmann/include")
  add_library(nlohmann_json::nlohmann_json ALIAS rmtc_json)
endif()
add_library(rmtc_core
  src/config.cpp src/perception.cpp src/memory.cpp src/planner.cpp src/policy.cpp
  src/combat.cpp src/runtime.cpp src/llm_client.cpp src/curl_transport.cpp
  src/policies/stage1_policy.cpp src/policies/stage2_policy.cpp)
target_include_directories(rmtc_core PUBLIC include)
target_link_libraries(rmtc_core PUBLIC rmtc_decision nlohmann_json::nlohmann_json Threads::Threads)
if(RMTC_WITH_CURL)
  find_package(CURL REQUIRED)
  target_link_libraries(rmtc_core PRIVATE CURL::libcurl)
  target_compile_definitions(rmtc_core PRIVATE RMTC_HAS_CURL=1)
endif()
if(MSVC)
  target_compile_options(rmtc_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(rmtc_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(rmtc_replay src/replay.cpp)
target_link_libraries(rmtc_replay PRIVATE rmtc_core)
if(BUILD_TESTING)
  add_executable(rmtc_core_tests test/test_core.cpp)
  target_link_libraries(rmtc_core_tests PRIVATE rmtc_core)
  target_compile_definitions(rmtc_core_tests PRIVATE RMTC_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  add_test(NAME bot_core_scenarios COMMAND rmtc_core_tests)
  add_executable(rmtc_multimap test/stage1_multimap.cpp)
  target_link_libraries(rmtc_multimap PRIVATE rmtc_core)
  add_test(NAME stage1_multimap COMMAND rmtc_multimap "${CMAKE_CURRENT_SOURCE_DIR}/test/maps/stage1_suite.json")
endif()
if(RMTC_WITH_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(std_msgs REQUIRED)
  add_executable(rmtc_bot src/main.cpp src/ros_io.cpp)
  # Humble's ament_target_dependencies uses the plain link signature.
  target_link_libraries(rmtc_bot rmtc_core)
  ament_target_dependencies(rmtc_bot rclcpp std_msgs)
  install(TARGETS rmtc_bot rmtc_replay DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
